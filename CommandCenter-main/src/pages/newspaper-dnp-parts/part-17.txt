/** Stable pager identity so later folio inserts (#350) do not remount A1. */
function folioSlotKey(page: { kind: string; folio: string } & Record<string, unknown>): string {
  switch (page.kind) {
    case "favorites-front":
      return "a-front";
    case "favorites-clubs":
      return `a-clubs-${String(page.weatherPart ?? "all")}-${String(page.clubOffset ?? 0)}`;
    case "favorites-form":
      return `a-form-${String(page.sectionPage ?? page.folio)}`;
    case "favorites-watch":
      return "a-watch";
    case "favorites-day":
      return "a-day";
    case "favorites-beez":
      return "a-beez";
    case "favorites-races":
      return `a-races-${String(page.sectionPage ?? 0)}`;
    case "favorites-continue": {
      const jumps = page.jumps as { card?: { id?: string } }[] | undefined;
      return `a-cont-${jumps?.[0]?.card?.id ?? page.folio}`;
    }
    case "favorites-inside": {
      const primary = page.primary as { id?: string } | undefined;
      return `a-in-${primary?.id ?? page.folio}`;
    }
    case "sport-front":
      return `s-${String(page.path)}-${String(page.focus)}-${String(page.sectionPage ?? 0)}`;
    case "sport-inside": {
      const primary = page.primary as { id?: string } | undefined;
      return `s-in-${primary?.id ?? page.folio}`;
    }
    case "national":
      return `n-${String(page.sectionPage ?? page.folio)}`;
    case "missouri":
      return `m-${String(page.sectionPage ?? page.folio)}`;
    default:
      return page.folio;
  }
}

/** A1 prints immediately. The rest fill backward through the folio, or as soon as you turn to them. */
function FolioSlot({
  index,
  folio,
  kind,
  render,
}: {
  index: number;
  folio: string;
  kind: string;
  render: () => ReactNode;
}) {
  const current = useContext(PagerIndexContext);
  const hydrate = useContext(FolioHydrateContext);
  const front = kind === "favorites-front" || index === 0;
  const near = front || index < hydrate || Math.abs(index - current) <= NEAR_PAGES;
  const [shown, setShown] = useState(front);
  useEffect(() => {
    if (near) setShown(true);
  }, [near]);
  return (
    <section className="wsj-page" aria-label={`Page ${folio}`} {...(near ? { "data-near": "" } : {})}>
      <div className="wsj-fit">
        <FittedSheet folio={folio} ready={shown}>{shown ? <FolioBody render={render} /> : null}</FittedSheet>
      </div>
    </section>
  );
}

function prefetchNearArt(pager: HTMLElement, index: number) {
  const sheets = pager.children;
  for (let i = 0; i < sheets.length; i++) {
    if (Math.abs(i - index) > NEAR_PAGES) continue;
    const sheet = sheets[i] as HTMLElement;
    for (const img of sheet.querySelectorAll("img")) prefetchSrc(img.currentSrc || img.src);
  }
}

/** Fetch the rest of the edition's art in idle time, a few at a time, so far folios open already printed. */
function warmEdition(pager: HTMLElement, cap = 4): () => void {
  const conn = (navigator as Navigator & { connection?: { saveData?: boolean; effectiveType?: string } }).connection;
  if (conn?.saveData || /(^|-)2g$/.test(conn?.effectiveType ?? "")) return () => {};
  const hasIdle = typeof window.requestIdleCallback === "function";
  const idle = (cb: () => void) => (hasIdle ? window.requestIdleCallback(cb, { timeout: 2_000 }) : window.setTimeout(cb, 200));
  const cancelIdle = (id: number) => (hasIdle ? window.cancelIdleCallback(id) : window.clearTimeout(id));
  let stopped = false;
  let inflight = 0;
  let handle = 0;
  const seen = new Set<string>();
  const pump = () => {
    handle = 0;
    if (stopped) return;
    for (const img of pager.querySelectorAll<HTMLImageElement>("img[src]")) {
      if (inflight >= cap) break;
      const src = img.currentSrc || img.src;
      if (!src || seen.has(src) || img.complete) continue;
      seen.add(src);
      prefetchSrc(src);
      inflight++;
      window.setTimeout(() => {
        inflight--;
        schedule();
      }, 800);
    }
  };
  const schedule = () => {
    if (!stopped && !handle) handle = idle(pump);
  };
  schedule();
  return () => {
    stopped = true;
    if (handle) cancelIdle(handle);
  };
}
